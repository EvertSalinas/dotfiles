# Arch/Omarchy equivalent of the Brewfile. One package per line, "#" lines are comments.
# Official: awk '/^\[official\]/{f=1;next} /^\[/{f=0} f&&!/^#/&&NF' packages-arch.txt | sudo pacman -S --needed -
# AUR:      awk '/^\[aur\]/{f=1;next} /^\[/{f=0} f&&!/^#/&&NF' packages-arch.txt | yay -S --needed -
